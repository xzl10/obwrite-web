ConductPresupposingPredicate
  := predicate whose satisfaction conditions quantify over
     the spatiotemporal trajectory of its subject

// PREMISE (DI from formal-physical-property-disjointness)
Subject ∈ FormalEntity
  ⇒ spatiotemporal trajectory ∉ AdmissiblePredicates(FormalEntity)

// THEOREM (presupposition failure; Strawson 1950)
ConductPresupposingPredicate(Subject) ∧ Subject ∈ FormalEntity
  ⇒ PresuppositionFailure(ConductPresupposingPredicate, Subject)
  ⇒ TruthValueGap: neither satisfied nor violated

// COROLLARY (performative contradiction vacuity; Austin 1962)

the utterer's spatiotemporal conduct
⇒ Referent ∈ FormalEntity ⇒ PerformativeContradiction(Referent) has no truth value
