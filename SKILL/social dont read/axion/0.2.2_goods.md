
[Hierarchy of Possibilities]
Constraint: All_Scenarios ⊃ Logically_Possible ⊃ Physically_Possible ⊃ Technically_Possible ⊃ Actual_World

~Logically_Possible := Logical_Impossibility (Error/Paradox)
Logically_Possible ∩ ~Physically_Possible := Physical_Impossibility (Meaning Test)
Physically_Possible ∩ ~Technically_Possible := Practical_Impossibility (Causality Test)

[Liebig's Law of the Minimum]
Constraint: Essential_Factors = { Factor_1, Factor_2, ..., Factor_n } (All_Non_Substitutable)
Satisfaction_Rate(Factor_i) := Supply(Factor_i) / Requirement(Factor_i)

Actual_Yield := Minimum{ Satisfaction_Rate(Factor_i) | Factor_i ∈ Essential_Factors }

Limiting_Factor := { Factor_i | Satisfaction_Rate(Factor_i) = Actual_Yield } → Bottleneck (Growth_Determinant)
Surplus_Factors := Essential_Factors \ Limiting_Factor → Ineffective_Excess (Zero_Marginal_Productivity)

Delta_Supply(Limiting_Factor) > 0 → Delta_Actual_Yield > 0 (Growth_Unlocked)
Delta_Supply(Surplus_Factors) > 0 → Delta_Actual_Yield = 0 (Barrel_Stave_Principle)

[Matrix of Economic Goods]
Private_Good := Excludable ∩ Rivalrous
Club_Good    := Excludable ∩ ~Rivalrous
Common_Good  := ~Excludable ∩ Rivalrous → Tragedy_of_Commons
Public_Good  := ~Excludable ∩ ~Rivalrous → Free_Rider_Problem
