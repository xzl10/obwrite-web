## Action

// SKILL := formal_physical_property_disjointness (stateless, project-agnostic)
// Pure function: EntityCategory × Predicate → Applicability. No side effects.

CORE := OntologicalCategory → AdmissiblePredicateSet → PredicateApplicability

// CATEGORIES (owned by modal-possibility-nesting; referenced, not redefined)
FormalEntity
  := entity whose identity conditions contain no spatial location or physical causal power

PhysicalEntity
  := entity whose identity conditions permit spatial location or physical causal power

// ADMISSIBLE PREDICATES (BFO type partition)
AdmissiblePredicates(FormalEntity)
  := { truth, falsity, validity, provability, consistency }

AdmissiblePredicates(PhysicalEntity)
  := { location, duration, mass, energy, causal interaction }

// THEOREM (property disjointness)
AdmissiblePredicates(FormalEntity)
  ∩ AdmissiblePredicates(PhysicalEntity)
  = ∅

// DEFINITION (applicability)
Applicable(Predicate, Entity)
  ⇔ Predicate ∈ AdmissiblePredicates(CategoryOf(Entity))

// DEFINITION (category mistake)
CategoryMistake(Predicate, Entity)
  ⇔ Predicate ∉ AdmissiblePredicates(CategoryOf(Entity))
