## Action

// SKILL := identification_by_removal (stateless), NOT explanation by accumulation

CORE := Overdetermined_Phenomenon → Monotone_Ablation → Irreducible_Core

// SETS
Components     := { c1, c2, ..., cn }
Configuration  ⊆ Components
Holds          : Configuration → { true, false }      // does the phenomenon still occur

// THEOREM (overdetermination blocks identification)
count{ Configuration ⊂ full : Holds(Configuration) } > 1 ⇔ Overdetermined
Overdetermined ⇒ observation of the full configuration identifies nothing
// several sufficient subsets at once ⇒ the cause is unidentifiable by observation

// THEOREM (necessity by removal)
Necessary(c) ⇔ ∀ Configuration: Holds(Configuration) → c ∈ Configuration
Core      := ⋂ { Configuration : Holds(Configuration) }
Amplifier := Components \ Core        // raises magnitude; carries no identification power

// INUS (Mackie 1974)
INUS(c) ⇔ insufficient alone ∧ necessary within a sufficient bundle
⇒ "not the sole cause" and "dispensable" are distinct verdicts

// PROCEDURE
order Configurations by strict inclusion ↦ monotone ladder; one removal per rung
Natural_Experiment := ablation produced by disturbance, not by the investigator
Limit_Case         := smallest Configuration with Holds = true ↦ maximal discriminating power
