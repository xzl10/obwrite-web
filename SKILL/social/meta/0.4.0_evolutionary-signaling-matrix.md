## Action

// SKILL := payoff_state_mapping (strict logic), NOT procedural branching
// SYSTEM := natural selection as a continuous mathematical function

CORE := Payoff × Information → [Evolution] → State × Dynamics

// AXIOMS & SETS (Space Definition)
ΔFitness = (ΔFitness(Sender), ΔFitness(Receiver)) ∈ {+, -}²
Information = Signal ∪ Cue ∪ Noise
Signal ∩ Cue = ∅

// THEOREM (The Evolutionary Intent Constraint)
// 送信者が利益を得ることは、その情報が意図的なシグナルであることの必要十分条件である。
ΔFitness(Sender) = + ⇔ x ∈ Signal
ΔFitness(Sender) = - ⇔ x ∈ Cue ∪ Noise

// MATRIX MAPPING: Map(ΔFitness(Sender), ΔFitness(Receiver)) ↦ (State, Dynamics)
Map(+, +) ↦ (Honest_Signal, ESS[Handicap ∪ Index])
Map(+, -) ↦ (Manipulation, Arms_Race)
Map(-, +) ↦ (Eavesdropping, Concealment_Pressure)
Map(-, -) ↦ (Spite_Ignoring, Extinction)
