## Action

// Pure function: Rule × OriginNodeSet × Interpretation → SelfApplicability. No side effects.
// Node ∈ { brain, server, LLM agent, organism, organization }: substrate-neutral

CORE := RuleScope × OriginNodeSet → NodeInclusion → SelfApplicability

// SETS
RuleScope(Rule, Interpretation)
  := { Node | PredicateOf(Rule) is true of Node under Interpretation }

OriginNodeSet(Rule)
  := { Node | Node asserts ∨ instantiates ∨ evaluates Rule }

// DEFINITION (origin-node inclusion)
RuleAppliesToAllOriginNodes(Rule, Interpretation)
  ⇔ OriginNodeSet(Rule) ⊆ RuleScope(Rule, Interpretation)

// DEFINITION (single-node self-applicability)
RuleAppliesToOriginNode(Node, Rule, Interpretation)
  ⇔ Node ∈ OriginNodeSet(Rule)
     ∧ Node ∈ RuleScope(Rule, Interpretation)

// THEOREM (universal instantiation / dictum de omni)
UniversalRule(Rule)
  ∧ Node ∈ RuleScope(Rule, Interpretation)
  ⇒ RuleAppliesToOriginNode(Node, Rule, Interpretation)
